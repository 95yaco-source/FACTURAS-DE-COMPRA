Respaldo Rápido e Importación Directa (Archivo .JSON)

Exporta toda tu base de datos (PDFs, vouchers de pago, directorio de proveedores y estado SIRE) en un solo archivo o importa el respaldo de otro equipo.

Importar Base (.JSON)

¿Cómo obtener tu Enlace de Sincronización gratuito con Google Drive?

1. Entra a script.google.com → Nuevo proyecto.
2. Pega el código de abajo y haz clic en Implementar → Nueva implementación → Aplicación web (Ejecutar como: Yo, Quién tiene acceso: Cualquier usuario).
3. Copia la URL de la aplicación web y pégala arriba en Enlace de Sincronización.

// Código Google Apps Script para Sincronizar Facturas PDF + Vouchers de Pago + Validación SIRE function doGet(e) { var folder = getOrCreateFolder_("FacturasCompra_SIRE"); var dbFile = getOrCreateDbFile_(folder); var data = JSON.parse(dbFile.getBlob().getDataAsString() || '{"invoices":[],"validation":null}'); return ContentService.createTextOutput(JSON.stringify({ ok: true, data: data })) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var payload = JSON.parse(e.postData.contents); var folder = getOrCreateFolder_("FacturasCompra_SIRE"); var dbFile = getOrCreateDbFile_(folder); var db = JSON.parse(dbFile.getBlob().getDataAsString() || '{"invoices":[],"validation":null}'); if (payload.action === "UPSERT_INVOICE" && payload.invoice) { var inv = payload.invoice; var idx = -1; for (var i = 0; i < db.invoices.length; i++) { if (db.invoices[i].normalizedKey === inv.normalizedKey) { idx = i; break; } } if (inv.pdfBase64) { var bytes = Utilities.base64Decode(inv.pdfBase64.split(",").pop()); var mime = inv.pdfBase64.indexOf("data:image") === 0 ? "image/jpeg" : "application/pdf"; var blob = Utilities.newBlob(bytes, mime, inv.fileName || (inv.normalizedKey + ".pdf")); var savedPdf = folder.createFile(blob); savedPdf.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); inv.driveFileId = savedPdf.getId(); delete inv.pdfBase64; } if (inv.voucherBase64 && inv.voucherBase64.indexOf("data:image") === 0) { var vMatch = inv.voucherBase64.match(/^data:(image\/[a-zA-Z0-9+.-]+);base64,/); var vMime = vMatch ? vMatch[1] : "image/jpeg"; var vExt = vMime.split("/")[1] || "jpg"; var vBytes = Utilities.base64Decode(inv.voucherBase64.split(",").pop()); var vBlob = Utilities.newBlob(vBytes, vMime, "VOUCHER_" + (inv.normalizedCode || inv.id) + "." + vExt); var savedVoucher = folder.createFile(vBlob); savedVoucher.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); inv.voucherDriveFileId = savedVoucher.getId(); delete inv.voucherBase64; } if (idx !== -1) { inv.driveFileId = inv.driveFileId || db.invoices[idx].driveFileId; if (inv.voucherBase64 === null) { delete inv.voucherDriveFileId; } else { inv.voucherDriveFileId = inv.voucherDriveFileId || db.invoices[idx].voucherDriveFileId; } db.invoices[idx] = inv; } else { db.invoices.push(inv); } } else if (payload.action === "DELETE_INVOICE" && payload.id) { db.invoices = db.invoices.filter(function(x) { return x.id !== payload.id; }); } else if (payload.action === "SAVE_VALIDATION") { db.validation = payload.validation; } else if (payload.action === "FULL_SYNC" && payload.data) { db = payload.data; } dbFile.setContent(JSON.stringify(db)); return ContentService.createTextOutput(JSON.stringify({ ok: true, data: db })) .setMimeType(ContentService.MimeType.JSON); } function getOrCreateFolder_(name) { var it = DriveApp.getFoldersByName(name); return it.hasNext() ? it.next() : DriveApp.createFolder(name); } function getOrCreateDbFile_(folder) { var it = folder.getFilesByName("database_compras_sire.json"); return it.hasNext() ? it.next() : folder.createFile("database_compras_sire.json", '{"invoices":[],"validation":null}', MimeType.PLAIN_TEXT); }
